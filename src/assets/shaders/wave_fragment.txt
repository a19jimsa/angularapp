#version 300 es
precision mediump float;
precision mediump sampler2DArray;

uniform sampler2DArray u_textures;
uniform sampler2D u_texture;
uniform float u_time;
uniform vec2 u_speed;

in vec2 v_texcoord;
in float v_age;

out vec4 FragColor;

void main() { 
    float opacity = v_age;
    vec2 uv = v_texcoord;
    vec2 uvScrolling = uv + vec2(u_speed.x * u_time, u_speed.y * u_time);
    vec4 baseColor = texture(u_textures, vec3(uvScrolling, 0));
    vec4 multColor = texture(u_texture, uv);
    float t = baseColor.r;
    vec3 color = mix(vec3(0.0, 0.0, 1.0), vec3(1.0), t);
    baseColor.rgb *= color;
    baseColor.rgb *= opacity;
    FragColor = vec4(baseColor.rgb, baseColor.r);
}

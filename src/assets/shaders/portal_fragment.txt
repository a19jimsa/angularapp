#version 300 es
precision mediump float;
precision mediump sampler2DArray;

uniform sampler2DArray u_textures;
uniform float u_time;

in vec2 v_texcoord;

out vec4 FragColor;

void main() { 
    vec2 uv = v_texcoord;
    vec4 baseColor = texture(u_textures, vec3(uv, 0));
    
    FragColor = baseColor;
}

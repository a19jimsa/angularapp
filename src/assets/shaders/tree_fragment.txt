#version 300 es
precision mediump float;
precision mediump sampler2DArray;

uniform sampler2DArray u_textures;
uniform float u_fogPower;
uniform vec3 u_fogColor;

in vec2 v_texcoord;
in float v_fogDistance;
in float v_index;

out vec4 FragColor;

void main() {
    vec4 treeTexture = texture(u_textures, vec3(v_texcoord, v_index));

    float fogFactor = smoothstep(0.0, 500.0 * u_fogPower, v_fogDistance);

    vec3 color = mix(treeTexture.rgb, u_fogColor, fogFactor);

    FragColor = vec4(color, treeTexture.a);
}

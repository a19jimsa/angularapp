#version 300 es
precision mediump float;

layout(location = 0) in vec3 a_position;
layout(location = 1) in vec2 a_texcoord;
layout(location = 2) in vec3 a_instancePosition;
layout(location = 3) in float a_size;
layout(location = 4) in float a_index;

uniform mat4 u_matrix;
uniform mat4 u_model;
uniform vec3 u_cameraPos;

out vec2 v_texcoord;
out float v_fogDistance;
out float v_index;

void main() {
   vec4 worldPos = u_model * vec4(a_position * a_size + a_instancePosition + vec3(0.0, a_size / 2.0 - 1.0, 0.0), 1.0);
   v_fogDistance = length(worldPos.xyz - u_cameraPos);
   v_texcoord = a_texcoord;
   v_index = a_index;
   gl_Position = u_matrix * worldPos;
}
